Actor Lexicon_NSP2_Chaingun : Weapon
{
	Weapon.SelectionOrder 700
	Weapon.AmmoUse 1
	Weapon.AmmoGive 20
	Weapon.AmmoType "Clip"
	Weapon.SlotNumber 4
	+WEAPON.CHEATNOTWEAPON
	Inventory.PickupMessage "$GOTCHAINGUN"
	Obituary "$OB_MPCHAINGUN"
	Tag "$TAG_CHAINGUN"
	States
	{
	Ready:
		CHGG A 1 A_WeaponReady
		Loop
	Deselect:
		CHGG A 1 A_Lower
		Loop
	Select:
		CHGG A 1 A_Raise
		Loop
	Fire:
		CHGG A 0 A_FireBullets(5.6,0,1,5)
		CHGG A 0 A_PlaySound("weapons/chngun",CHAN_WEAPON)
		CHGG A 2 A_GunFlash
		CHGG B 0 A_FireBullets(5.6,0,1,5)
		CHGG B 0 A_PlaySound("weapons/chngun",CHAN_WEAPON)
		CHGG B 2 A_GunFlash("Flash2")
		CHGG B 0 A_ReFire
		Goto Ready
	Flash:
		CHGF A 3 Bright A_Light1
		Goto LightDone
	Flash2:
		SHTG E 0 Bright A_Light0
		Goto LightDone
	Spawn:
		MGUN A -1
		Stop
	}
}
